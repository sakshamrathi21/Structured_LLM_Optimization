\section{Introduction}
\begin{frame}{Motivation}
    \begin{itemize}
        \item Many users share the same prefix/system prompt\pause
        \item Example 1: OS TA Chatbot with shared lecture notes\pause
        \item Example 2: Enterprise assistants over shared product docs \pause
        \item Recomputing attention over identical prefixes = waste \pause
        \item Large batch sizes $\rightarrow$ exploding KV-cache memory usage
    \end{itemize}
\end{frame}

\begin{frame}{Problem Statement}
    \begin{itemize}
        \item \textbf{Goal:} Characterize LLM inference performance with shared prefixes \pause
        \item Identify key GPU-level performance metrics \pause
        \item Examine shared-prefix optimizations in inference engines \pause
        \item Study effects of prefix caching, chunked prefill, scheduling \pause
        \item Understand scalability bottlenecks with many prefix families
    \end{itemize}
\end{frame}

\section{Background}

\begin{frame}{Prefill vs Decode}
\begin{itemize}
    \item \textbf{Prefill:} Takes the whole user prompt as input and computes the probability of the first new token. Generate key and value vectors. Can be parallelized because all the tokens are known. \pause
    \item \textbf{Decode:} Generates the remaining new tokens sequentially. Stops when the maximum length is reached, or when an end-of-sequence token is emitted. Autoregressive in nature.
\end{itemize}
\end{frame}

\begin{frame}{Inference Engines}

\end{frame}



\section{Prior Work}


\begin{frame}{vLLM}
    \begin{columns}[T, totalwidth=\textwidth]
        \begin{column}{0.4\textwidth}
            \vspace{1.5cm}
            \centering
            \includegraphics[width=\linewidth]{\FIGDIR/vllm_system_overview.png}
            \vspace{0.3em}
            \captionof{figure}{vLLM system overview}
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item Centralized scheduler for coordinating GPU workers \pause
        \item \textbf{PagedAttention:} splits KV cache into fixed-size blocks \pause
        \item Eliminates internal/external fragmentation \pause
        \item Shared blocks for identical prefixes / beam search (copy-on-write)  \pause
        \item Evicted blocks moved to CPU memory \pause
        \item Each block tracked via a hash (parent hash + tokens + metadata)
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{FastTree}
    \begin{columns}[T, totalwidth=\textwidth]
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{\FIGDIR/kv_cache_sharing_through_radix_tree.png}
            \vspace{0.3em}
            \captionof{figure}{KV Cache Sharing through RadixTree}
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item GPU kernels operate over shared prefixes via a radix tree \pause
                \item Optimal edge assignment minimizes tile padding + intermediate-state loss \pause
                \item Merges same-prefix queries into one kernel $\rightarrow$ fewer memory accesses, GEMV $\rightarrow$ GEMM
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{BatchLLM}
\begin{itemize}
    \item Accelerates large-batch LLM inference\pause
    \item \textbf{Global Prefix Sharing:} Build prefix tree \emph{ahead-of-time}; avoids LRU eviction; near-optimal KV reuse \pause
    \item \textbf{Group Scheduling:} Execute requests as prefix-sharing groups; compute prefix once; free KV early \pause
    \item Reorder groups by decode/prefill ratio
\end{itemize}
\end{frame}


\begin{frame}{What's Missing?}
\begin{itemize}
    \item Current focus is primarily on small-request patterns with RadixTree-style structures\pause
    \item Missing discussion of ``family'' workloads — multiple requests sharing a large prefix\pause
    \item Missing analysis of working-set size and L2 hit rate, with too much emphasis on HBM bandwidth
\end{itemize}
\end{frame}

    \begin{itemize}

    \item Small homogeneous batches can outperform larger heterogeneous ones.

\section{Future Steps}
