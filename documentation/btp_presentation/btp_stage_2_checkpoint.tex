\section{Motivation}
\section{Motivation}
\begin{frame}{Motivation}
  \begin{itemize}
    \item Modern LLM inference is often \textbf{memory-bandwidth bound} during decode
    \item KV-cache accesses dominate performance for long-context workloads
    \item Existing systems primarily optimize \textbf{HBM capacity utilization}
    \item However, decode efficiency also depends on \textbf{temporal and spatial locality}
    \item Current schedulers do not explicitly account for \textbf{cache behavior or DRAM bandwidth utilization}
  \end{itemize}
\end{frame}

\begin{frame}{Core Scheduling Problem}
  \begin{itemize}
    \item Multiple requests often share long token prefixes
    \item During decode, batches issue simultaneous KV-cache memory requests
    \item Heterogeneous batches reduce:
      \begin{itemize}
        \item Cache hit rates
        \item DRAM access efficiency
        \item Effective bandwidth utilization
      \end{itemize}
    \item Fully homogeneous batches improve locality but may hurt system-level efficiency
  \end{itemize}
  \begin{block}{Goal}
    Construct batches that balance GPU utilization and prefix locality
  \end{block}
\end{frame}

\section{Key Idea}
\begin{frame}{Key Conclusions from the Initial Experiments}
  \begin{itemize}
    \item Decode-phase memory requests from batched requests are issued nearly \textbf{simultaneously}
    \item \textbf{Sequential} KV accesses improve effective DRAM bandwidth utilization
    \item Shared-prefix batches increase \textbf{temporal} and \textbf{spatial} locality
    \item Batches should be completely \textbf{homogeneous} to utilize these benefits fully
    \item \textbf{Greater prefix overlap} across active requests improves cache hit rates
    \item Merging requests from the same prefix together can also help in efficient KV cache eviction from the HBM $\rightarrow$ \textbf{avoiding recomputation}
  \end{itemize}
\end{frame}


\begin{frame}{Key Metric and Direction}

\textbf{Objective:}
\begin{itemize}
    \item Maximize concurrent GPU utilization
    \item Maximize shared-prefix locality duration
\end{itemize}
\vspace{0.2cm}

\textbf{Metric:}
\vspace{-0.2cm}
\[
(\text{Batch Size}, \text{Shared Prefix Chunks (common to \textbf{all} active requests)})
\]
\vspace{-0.7cm}
\begin{itemize}
    \item Larger batch size $\rightarrow$ higher GPU utilization
    \item More shared chunks $\rightarrow$ sustained temporal and spatial locality
    \item Only chunks shared by \textbf{all} requests contribute to consistent reuse
\end{itemize}

\begin{block}{Design Questions:}
    \begin{itemize}
    \item How do we efficiently compute shared-prefix overlap at scale? 
          $\rightarrow$ \textbf{Chunked Hash Tree}
    \item When should we stop adding requests to a batch?
          $\rightarrow$ \textbf{Reinforcement Learning (RL) Decision Maker}
\end{itemize}
\end{block}

\end{frame}


\section{Chunked Hash Tree}

\begin{frame}
    \Huge{\centerline{\bf Chunked Hash Tree}}
\end{frame}

\begin{frame}{Why We Need a New Data Structure}
\begin{itemize}
    \item Scheduling requires computing: $\text{Shared Chunks Across Active Requests}$
    \item Naively comparing token prefixes is:
    \begin{itemize}
        \item Too slow for long contexts (10K+ tokens)
        \item Too expensive at high request rates
    \end{itemize}
    \item Radix trees:
    \begin{itemize}
        \item Require node splits
        \item Expensive pointer traversal
    \end{itemize}
\end{itemize}

\begin{block}{Requirement}
We need a faster overlap computation  without explicit tree traversal.
\end{block}

\end{frame}

\begin{frame}{Chunked Hash Tree}

\begin{block}{Core Idea}
Convert prefix-overlap computation into hash-set intersection over chunked prefixes.
\end{block}

\begin{itemize}
    \item Divide tokens into fixed-size chunks of size $c$
    
    \item For each request, compute cumulative prefix hashes: $h_l = hash(tokens[0 : l \cdot c])$
    % \[
    %     h_l = \texttt{hash(tokens[0 : l \cdot c])}
    % \]
    
    \item Represent each request as a set of \textbf{(level, hash)} pairs
    
    \item Working Set:
    \[
        W = \{(l, h_l)\ \text{for all active requests}\}
    \]
    
    \item Shared chunks = levels where all active requests share the same hash
    
    \item Overlap computation reduces to counting shared hashes
    (no tree traversal, no node splitting)
\end{itemize}

\end{frame}

\section{RL Decision Maker}

\begin{frame}
    \Huge{\centerline{\bf RL Decision Maker}}
\end{frame}
\begin{frame}{Formulation}
    \begin{block}{Tradeoff}
        Add more requests to increase GPU utilization, but risk reducing shared-prefix locality.
    \end{block}
  \begin{block}{State}
        \begin{itemize}
        \item \textbf{B:} current batch size
        \item $\mathbf{C_B:}$ shared chunks across current active batch
        \item $\mathbf{C_A:}$ shared chunks if best candidate is added
        \item $\mathbf{R_W:}$ number of requests waiting at the end of $C_A$
    \end{itemize}
    \end{block}
\end{frame}


\begin{frame}{Formulation}
  \begin{block}{Action}
        \begin{itemize}
        \item \textbf{ADD:} pop and activate the best request from the chunked hash tree
        \item \textbf{SKIP:} break the batch construction loop, dispatch the batch to GPU
    \end{itemize}
    \end{block}
    \begin{block}{Reward}
        \begin{itemize}
        \item Negative decode wall step time of the previous batch
        \item Delivered once per batch
    \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Policies}
  \begin{itemize}
    \item \textbf{Q Learning:} Learns when sacrificing locality now may improve long-term throughput across future batches.
    \item \textbf{Contextual Bandit:} Optimizes immediate throughput of the next batch only.
    \item \textbf{Heuristics:} Static thresholds on batch size or chunk drop.
  \end{itemize}

\end{frame}
