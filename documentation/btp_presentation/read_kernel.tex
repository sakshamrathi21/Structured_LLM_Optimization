\section{Simplified Read Kernel}

\begin{frame}{Simplified Read Kernel}
    \begin{itemize}
        \item \textbf{Goal:} Model family-workload by kernels that have the sole job of reading KV cache blocks from the HBM
        \item Each request reads a huge family-shared prefix and a distinct suffix
        \item Each kernel launch creates its own grid
        \item All kernels submitted to a single stream
    \end{itemize}
\end{frame}

\begin{frame}{Block-to-Request Mapping}
\textbf{Definitions:}
\begin{itemize}
    \item Blocks per request = $B$
    \item Total blocks = requests $\times B$
\end{itemize}

\vspace{0.4cm}

\textbf{Inside the kernel:}
\begin{itemize}
    \item The request id can be found as: blockIdx.x / B
    \item The corresponding block in the rqeuest can be found as: blockIdx.x \% B
\end{itemize}

\vspace{0.4cm}

\textbf{Effect:}
\begin{itemize}
    \item Blocks from different requests execute concurrently
    \item GPU scheduler interleaves them naturally
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Memory Structure}
\begin{verbatim}
[d_mem]
 |-- Family 0
 |    |-- Large Shared Prefix
 |    |-- Request 0 Suffix
 |    |-- Request 1 Suffix
 |    |-- ...
 |-- Family 1
 |    |-- ...

\end{verbatim}

\end{frame}


\begin{frame}{Optimizations}
    \textbf{CUDA Graph Integration:}
\begin{itemize}
    \item Single kernel launch captured in graph
    \item Graph instantiated once
    \item Launched repeatedly (corresponding to multiple decode iterations)
\end{itemize}

\textbf{Loop Unrolling:}
\begin{itemize}
    \item Unroll inner loop reading block data
    \item Each thread reads multiple elements (in a single for loop iteration)
\end{itemize}
\end{frame}


\begin{frame}{DRAM Accesses' Statistics for vLLM}
    \begin{itemize}
        \item \textbf{DRAMA:} A DCGMI metric that counts the fraction of cycles the DRAM pipe is active in a unit time interval
        \item A higher instantaneous value indicates more DRAM activity/utilization
        \item We sample this metric every 1ms during the kernel execution
        \item We run two experiments (all in the same batch):
        \begin{itemize}
            \item Single family with 500 requests
            \item 5 families with 100 requests each
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{DRAM Accesses' Statistics for vLLM}
\begin{itemize}
        \item For the first case, the median DRAM utilization was found to be 0.106, whereas for the second case, it was 0.02, roughly 5x lesser
        \item Homogeneous batches $\rightarrow$ Higher DRAM utilization 
        \item If we sum over the DRAM utilization samples, we find that the single family case has a sum of 2677.81, whereas the 5 family case has a sum of 3781.25. If we assume that the amount of data transferred is proportional to the sum of DRAM utilizations, we can say that the 5 family case transfers 1.41x more data than the single family case
        \item This in turn gives us a hint that we are getting higher L2 cache hit rates in the single-family case
        \item The SM Occupancy was around 0.40 for both the cases
\end{itemize}
\end{frame}
